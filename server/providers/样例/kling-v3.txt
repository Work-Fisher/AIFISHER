
文生视频

  --url https://api-beijing.klingai.com/v1/videos/text2video \
  --header 'Authorization: Bearer <token>' \
  --header 'Content-Type: application/json' \
  --data '{
    "model_name": "kling-v3",
    "prompt": "一只可爱的小兔子，戴着眼镜，坐在桌边，看报纸，桌上放着一杯卡布奇诺",
    "negative_prompt": "",
    "mode": "pro",
    "sound": "on",
    "duration": "5",
    "aspect_ratio": "16:9",
    "watermark_info": "false"
  }'
